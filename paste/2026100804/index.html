using namespace std;
int dep[200005],son[200005],top[200005],ge[200005],fat[200005],id[200005],id2[200005],n,m,s,mod,cnt;
int a[200005];
vector<int>ve[200005];
void dfs(int x,int fa)
{
    dep[x] = dep[fa] + 1,ge[x] = 1,fat[x] = fa,son[x] = n + 1;
    for(int i = 0;i < ve[x].size();i ++)
    {
        dfs(ve[x][i],x);
        ge[x] += ge[ve[x][i]];
        if(ge[son[x]] < ge[ve[x][i]])son[x] = ve[x][i];
    }
    if(son[x] == n + 1)son[x] = 0;
}
void dfs2(int x,int y)
{
    top[x] = y,id[x] = ++cnt,id2[cnt] = a[x];
    //cout << x << " " << cnt << " " << id[x] << " " << son[x] << '\n';
    if(!son[x])return ;
    dfs2(son[x],y);
    for(int i = 0;i < ve[x].size();i ++)
    {
        if(ve[x][i] == son[x])continue ;
        dfs2(ve[x][i],ve[x][i]);
    }
}
struct node
{
    int ze,ye,sum,add;
}tr[800005];
void pushup(int p)
{
    tr[p].sum = tr[p * 2 + 1].sum + tr[p * 2 + 2].sum;
}
void pushdown(int p)
{
    if(tr[p].add == 1)
    {
        tr[p * 2 + 1].sum = (tr[p * 2 + 1].ye - tr[p * 2 + 1].ze + 1);
        tr[p * 2 + 2].sum = (tr[p * 2 + 2].ye - tr[p * 2 + 2].ze + 1);
        tr[p * 2 + 1].add = tr[p].add;
        tr[p * 2 + 2].add = tr[p].add;
    }
    if(tr[p].add == -1)
    {
        tr[p * 2 + 1].sum = 0;
        tr[p * 2 + 2].sum = 0;
        tr[p * 2 + 1].add = tr[p].add;
        tr[p * 2 + 2].add = tr[p].add;
    }
	tr[p].add = 0;
}
void build(int p,int x,int y)
{
    tr[p] = {x,y,0,0};
    if(x == y){return ;}
    build(p * 2 + 1,x,(x + y) >> 1);
    build(p * 2 + 2,((x + y) >> 1) + 1,y);
    pushup(p);
}
void jia(int x,int u,int v,int k)
{
    pushdown(x);
	if(tr[x].ze == u && v == tr[x].ye)
	{
		if(k == 1)tr[x].sum = (v - u + 1) * k,tr[x].add = k;
		else tr[x].sum = 0,tr[x].add = -1;
		return ;
	}
	int mid = (tr[x].ze + tr[x].ye) >> 1;
	if(u <= mid)jia(x * 2 + 1,u,min(v,mid),k);
	if(v > mid)jia(x * 2 + 2,max(u,mid + 1),v,k);
	pushup(x);
}
int query(int x,int u,int v)
{
    pushdown(x);
	int num = 0;
	if(tr[x].ze == u && v == tr[x].ye)return tr[x].sum;
	int mid = (tr[x].ze + tr[x].ye) >> 1;
	if(u <= mid)num += query(x * 2 + 1,u,min(v,mid));
	if(v > mid)num += query(x * 2 + 2,max(u,mid + 1),v);
	return num;
}
void lca(int u,int v)
{
    while(top[u] != top[v])
    {
        if(dep[top[u]] < dep[top[v]])swap(u,v);
        jia(0,id[top[u]],id[u],1);
        u = fat[top[u]];
    }
    if(dep[u] < dep[v])jia(0,id[u],id[v],1);
    else jia(0,id[v],id[u],1);
}
int lca2(int u,int v)
{
    int res = 0;
    while(top[u] != top[v])
    {
        if(dep[top[u]] < dep[top[v]])swap(u,v);
       // cout << id[top[u]] << " " << id[u] <<query(0,id[top[u]],id[u]) << '\n';
        res += query(0,id[top[u]],id[u]);
        u = fat[top[u]];
    }
   // cout << id[u] << " " << v << " " << id[v] << " " << query(0,id[u],id[v]) << '\n';
    if(dep[u] < dep[v])res += query(0,id[u],id[v]);
    else res += query(0,id[v],id[u]);
    return res;
}
void gen(int x)
{
    jia(0,id[x],id[x] + ge[x] - 1,0);
}
int gen2(int x)
{
    return query(0,id[x],id[x] + ge[x] - 1);
}
