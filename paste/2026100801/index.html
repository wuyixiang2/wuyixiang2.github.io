#include <iostream>
#include <vector>
#define int long long
using namespace std;
int dep[200005],son[200005],top[200005],ge[200005],fat[200005],id[200005],n,m,cnt;
int a[200005],zu[200005],yo[200005],ch[200005];
vector<int>ve[200005];
void dfs(int x,int fa)
{
    dep[x] = dep[fa] + 1,ge[x] = 1,fat[x] = fa;
    for(int i = 0;i < ve[x].size();i ++)
    {
        if(ve[x][i] == fa)continue ;
        dfs(ve[x][i],x);
        ge[x] += ge[ve[x][i]];
        if(ge[son[x]] < ge[ve[x][i]])son[x] = ve[x][i];
    }
}
void dfs2(int x,int y)
{
    top[x] = y,id[x] = ++cnt;
    if(!son[x])return ;
    dfs2(son[x],y);
    for(int i = 0;i < ve[x].size();i ++)
    {
        if(ve[x][i] == son[x] || ve[x][i] == fat[x])continue ;
        dfs2(ve[x][i],ve[x][i]);
    }
}
struct node
{
    int ze,ye,sum,maxx,minn,add;
}tr[800005];
void pushup(int p)
{
    tr[p].sum = tr[p * 2 + 1].sum + tr[p * 2 + 2].sum;
    tr[p].minn = min(tr[p * 2 + 1].minn,tr[p * 2 + 2].minn);
    tr[p].maxx = max(tr[p * 2 + 1].maxx,tr[p * 2 + 2].maxx);
}
void pushdown(int p)
{
    if(tr[p].add)
    {
        tr[p * 2 + 1].sum = -tr[p * 2 + 1].sum,tr[p * 2 + 2].sum = -tr[p * 2 + 2].sum;
        int mx = tr[p * 2 + 1].maxx,mn = tr[p * 2 + 1].minn;
        tr[p * 2 + 1].maxx = -mn,tr[p * 2 + 1].minn = -mx;
        mx = tr[p * 2 + 2].maxx,mn = tr[p * 2 + 2].minn;
        tr[p * 2 + 2].maxx = -mn,tr[p * 2 + 2].minn = -mx;
        tr[p * 2 + 1].add ^= tr[p].add,tr[p * 2 + 2].add ^= tr[p].add;
        tr[p].add = 0;
    }
}
void build(int p,int x,int y)
{
    tr[p] = {x,y,0,0,0,0};
    if(x == y){return ;}
    build(p * 2 + 1,x,(x + y) >> 1);
    build(p * 2 + 2,((x + y) >> 1) + 1,y);
    pushup(p);
}
void jia(int x,int u,int v)
{
	if(tr[x].ze == u && v == tr[x].ye)
	{
        tr[x].add ^= 1,tr[x].sum *= -1;
        int mx = tr[x].maxx,mn = tr[x].minn;
        tr[x].maxx = -mn,tr[x].minn = -mx;
		return ;
	}
    pushdown(x);
	int mid = (tr[x].ze + tr[x].ye) >> 1;
	if(u <= mid)jia(x * 2 + 1,u,min(v,mid));
	if(v > mid)jia(x * 2 + 2,max(u,mid + 1),v);
	pushup(x);
}
void cover(int x,int u,int v,int k)
{
	if(tr[x].ze == u && v == tr[x].ye)
	{
		tr[x].sum = tr[x].maxx = tr[x].minn = k;
        tr[x].add = 0;
		return ;
	}
    pushdown(x);
	int mid = (tr[x].ze + tr[x].ye) >> 1;
	if(u <= mid)cover(x * 2 + 1,u,min(v,mid),k);
	if(v > mid)cover(x * 2 + 2,max(u,mid + 1),v,k);
	pushup(x);
}
int query(int x,int u,int v)
{
    pushdown(x);
	int num = -1e9;
	if(tr[x].ze == u && v == tr[x].ye)return tr[x].maxx;
	int mid = (tr[x].ze + tr[x].ye) >> 1;
	if(u <= mid)num = max(num,query(x * 2 + 1,u,min(v,mid)));
	if(v > mid)num = max(num,query(x * 2 + 2,max(u,mid + 1),v));
	return num;
}
int query2(int x,int u,int v)
{
    pushdown(x);
	int num = 1e9;
	if(tr[x].ze == u && v == tr[x].ye)return tr[x].minn;
	int mid = (tr[x].ze + tr[x].ye) >> 1;
	if(u <= mid)num = min(num,query2(x * 2 + 1,u,min(v,mid)));
	if(v > mid)num = min(num,query2(x * 2 + 2,max(u,mid + 1),v));
	return num;
}
int query3(int x,int u,int v)
{
    pushdown(x);
	int num = 0;
	if(tr[x].ze == u && v == tr[x].ye)return tr[x].sum;
	int mid = (tr[x].ze + tr[x].ye) >> 1;
	if(u <= mid)num += query3(x * 2 + 1,u,min(v,mid));
	if(v > mid)num += query3(x * 2 + 2,max(u,mid + 1),v);
	return num;
}
void add(int u,int v)
{
    while(top[u] != top[v])
    {
        if(dep[top[u]] < dep[top[v]])swap(u,v);
        jia(0,id[top[u]],id[u]);
        u = fat[top[u]];
    }
    if(u == v)return ;
    if(dep[u] < dep[v])jia(0,id[son[u]],id[v]);
    else jia(0,id[son[v]],id[u]);
}
void change(int u,int v,int zh)
{
    while(top[u] != top[v])
    {
        if(dep[top[u]] < dep[top[v]])swap(u,v);
        cover(0,id[top[u]],id[u],zh);
        u = fat[top[u]];
    }
    if(u == v)return ;
    if(dep[u] < dep[v])cover(0,id[son[u]],id[v],zh);
    else cover(0,id[son[v]],id[u],zh);
}
int lca2(int u,int v)
{
    int res = -1e9;
    while(top[u] != top[v])
    {
        if(dep[top[u]] < dep[top[v]])swap(u,v);
        res = max(res,query(0,id[top[u]],id[u]));
        u = fat[top[u]];
    }
    if(u == v)return res;
    if(dep[u] < dep[v])res = max(res,query(0,id[son[u]],id[v]));
    else res = max(res,query(0,id[son[v]],id[u]));
    return res;
}
int lca3(int u,int v)
{
    int res = 1e9;
    while(top[u] != top[v])
    {
        if(dep[top[u]] < dep[top[v]])swap(u,v);
        res = min(res,query2(0,id[top[u]],id[u]));
        u = fat[top[u]];
    }
    if(u == v)return res;
    if(dep[u] < dep[v])res = min(res,query2(0,id[son[u]],id[v]));
    else res = min(res,query2(0,id[son[v]],id[u]));
    return res;
}
int lca4(int u,int v)
{
    int res = 0;
    while(top[u] != top[v])
    {
        if(dep[top[u]] < dep[top[v]])swap(u,v);
        res += query3(0,id[top[u]],id[u]);
        u = fat[top[u]];
    }
    if(u == v)return res;
    if(dep[u] < dep[v])res += query3(0,id[son[u]],id[v]);
    else res += query3(0,id[son[v]],id[u]);
    return res;
}
signed main()
{
    ios::sync_with_stdio(0);
    cin.tie(0),cout.tie(0);
    cin >> n;
    for(int i = 1;i < n;i ++)
    {
        int x,y;
        cin >> x >> y,x ++,y ++;
        ve[x].push_back(y),ve[y].push_back(x);
        cin >> ch[i];zu[i] = x,yo[i] = y;
	}
    dfs(1,0);
    dfs2(1,1);
    build(0,1,n);
    for(int i = 1;i < n;i ++)change(zu[i],yo[i],ch[i]);
    cin >> m;
    for(int i = 1,y,z,zh;i <= m;i ++)
    {
        string x;
        cin >> x;
        cin >> y >> z;
        if(x != "C")y ++,z ++;
        if(x == "C")change(zu[y],yo[y],z);
        else if(x == "N")add(y,z);
        else if(x == "MAX")cout << lca2(y,z) << '\n';
        else if(x == "MIN")cout << lca3(y,z) << '\n';
        else cout << lca4(y,z) << '\n';
    }
}
