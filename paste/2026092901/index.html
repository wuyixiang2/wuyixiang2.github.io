#include <iostream>
#include <cmath>
#include <map>
using namespace std;
int n,m,add;
double a[105][105],b[105],x[105],w[105];
map<string,int>mp;
void GaoSi()
{
    for(int i = 1;i <= n;i ++)b[i] = 0;
    for(int i = 1;i <= m;i ++)
    {
        int r = 0;
        for(int k = i;k <= n;k ++)
			if(fabs(a[k][i]) > 0)//不为0，则交换
			{
				swap(a[i],a[k]);
				swap(b[i],b[k]);
				break ;
			}
		if(a[i][i] == 0)continue ;
		for(int j = 1;j < n;j ++)
		{
		    if(j + 1 == i)continue ;
			double wj = (a[j + 1][i] / a[i][i]);//加减消元
			for(int k = i;k <= m;k ++)a[j + 1][k] -= a[i][k] * wj;
			b[j + 1] -= b[i] * wj;
		}
    }
    int ans = 0;
    for(int i = 1;i <= n;i ++)
    {
        for(int j = 1;j <= m;j ++)
            if(fabs(a[i][j]) > 0.00001)
            {
                ans ++;
                break ;
            }
    }
    if(ans < m)cout << "Y\n";
    else cout << "N\n";
}
void solve()
{
    cin >> m;n = 0;
    mp.clear();
    for(int i = 1;i <= m;i ++)
    {
        string s;
        cin >> s;
        int mo = 0;
        for(int j = 0;j < s.size();j ++)
        {
            if(s[j] >= '0' && s[j] <= '9')
            {
                int k = j,ccnt = s[j] - '0';
                while(k + 1 < s.size() && s[k + 1] >= '0' && s[k + 1] <= '9')k ++,ccnt = ccnt * 10 + s[k] - '0';
                string t = s.substr(mo,j - mo);
                if(!mp[t])mp[t] = ++n;
                a[mp[t]][i] = ccnt;
                mo = k + 1,j = k;
            }
        }
    }/*
    for(int i = 1;i <= n;i ++)
    {
        for(int j = 1;j <= m;j ++)cout << a[i][j] << " ";
        cout << '\n';
    }*/
    GaoSi();
    for(int i = 1;i <= n;i ++)
        for(int j = 1;j <= m;j ++)a[i][j] = 0;
}
int main()
{
    int T;
    cin >> T;
    while(T --)solve();
}
